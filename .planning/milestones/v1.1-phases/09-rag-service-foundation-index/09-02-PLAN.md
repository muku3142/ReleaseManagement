// domain
public record ClientApp(UUID id, String name, Instant createdAt, Instant revokedAt, Instant lastUsedAt) {}
public record DocumentCollection(UUID id, UUID clientId, String name, String embeddingModel, Instant createdAt) {}
public record CollectionStatus(String name, String embeddingModel, long pending, long indexed, long failed, Instant lastIndexedAt) {}
public enum DocumentStatus { PENDING, INDEXED, FAILED }
public record DocumentContent(String externalId, String title, String text, String url, Map<String, Object> metadata) {}
public record DocumentView(String externalId, String title, String url, Map<String, Object> metadata,
                           DocumentStatus status, String statusReason, Instant indexedAt, Instant updatedAt) {}
public record ClaimedDocument(UUID id, UUID collectionId, String embeddingModel, String title, String text, byte[] contentHash) {}
public record ChunkEmbedding(int ordinal, String content, float[] embedding) {}
public record MetadataFilter(List<Clause> clauses) {
    public record Clause(String key, List<Object> anyOf) {}   // anyOf.size()==1 is equality, >1 is an `in` list
    public static MetadataFilter none() { return new MetadataFilter(List.of()); }
}
public record SearchHit(String documentId, String title, String url, Map<String, Object> metadata,
                        int chunkIndex, String text, double score) {}
public record QueueDepth(long pending, long failed) {}
public enum EmbeddingPurpose { DOCUMENT, QUERY }   // callers state intent; the embedding adapter maps it to its model's convention (nomic prefix, Cohere input_type, ...)

// domain exceptions: all `extends RuntimeException`, constructor takes a message that never contains document text, titles,
// queries or keys (D-locked: logs record metadata only)
CollectionNotFoundException(String collectionName)    // 404 COLLECTION_NOT_FOUND
DocumentNotFoundException(String externalId)          // 404 DOCUMENT_NOT_FOUND
ClientNotFoundException(UUID clientId)                // 404 CLIENT_NOT_FOUND
ClientNameTakenException(String name)                 // 409 CLIENT_NAME_TAKEN
EmbeddingModelNotAllowedException(String model)       // 400 MODEL_NOT_ALLOWED
CollectionModelConflictException(String collectionName) // 409 COLLECTION_MODEL_CONFLICT
PayloadTooLargeException(long limitBytes)             // 413 PAYLOAD_TOO_LARGE
InvalidContentException(String message)               // 400 INVALID_CONTENT (message names the field/rule, never echoes content)
EmbeddingUnavailableException(String message, Throwable cause) // outage; search -> 503 EMBEDDING_UNAVAILABLE
EmbeddingRejectedException(String message)            // content error; search -> 400 QUERY_REJECTED

// ports
public interface ClientStore {
    Optional<ClientApp> create(String name, byte[] keyHash);        // empty when the name is taken
    Optional<UUID> findActiveIdByKeyHash(byte[] keyHash);           // revoked keys never match
    void touchLastUsed(UUID clientId);                              // at most one write per minute per client (D-09-04)
    boolean replaceKey(UUID clientId, byte[] keyHash);              // false when unknown or revoked
    boolean revoke(UUID clientId);                                  // false when unknown
    Optional<ClientApp> find(UUID clientId);
    List<ClientApp> list();
}
public interface CollectionStore {
    Optional<DocumentCollection> find(UUID clientId, String name);
    Optional<DocumentCollection> create(UUID clientId, String name, String embeddingModel); // empty on (client_id,name) conflict
    List<CollectionStatus> listStatus(UUID clientId);
    Optional<CollectionStatus> findStatus(UUID clientId, String name);
}
public interface DocumentStore {
    DocumentStatus upsert(UUID collectionId, DocumentContent content, byte[] contentHash); // returns resulting status
    boolean delete(UUID collectionId, String externalId);
    Optional<DocumentView> find(UUID collectionId, String externalId);
    int requeueAll(UUID collectionId);
    QueueDepth queueDepth(UUID clientId);
}
public interface IndexQueue {
    List<ClaimedDocument> claim(int batchSize, Duration lease);
    void release(Collection<UUID> documentIds);
    boolean complete(UUID documentId, byte[] contentHash, String embeddingModel, List<ChunkEmbedding> chunks); // false = stale, discarded
    void fail(UUID documentId, byte[] contentHash, String reason);
}
public interface ChunkSearch {
    List<SearchHit> search(UUID collectionId, String embeddingModel, float[] queryVector, MetadataFilter filter, int k);
}
public interface EmbeddingPort {
    int DIMENSIONS = 768;   // must equal vector(768) in V1__rag_schema.sql
    List<float[]> embed(String model, EmbeddingPurpose purpose, List<String> inputs); // throws EmbeddingUnavailableException / EmbeddingRejectedException
    boolean isAvailable();
}
