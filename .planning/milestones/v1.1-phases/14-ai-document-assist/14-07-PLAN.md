```
GET  /api/assistant/transform/limits -> {"numCtx":4096,"charsPerToken":3.0,"systemTokens":152,"templateSlack":256,
       "instructionMaxChars":2000,"presets":{"CLEAN_UP":{"RELEASE_NOTES":5325,"RUNBOOK":5277,"APPROVAL":5262},
       "SUMMARIZE":{"RELEASE_NOTES":5406,"RUNBOOK":5388,"APPROVAL":5367}}}         (numbers follow the configured texts)
POST /api/assistant/transform  {"releaseId":uuid,"documentType":"RELEASE_NOTES"|"RUNBOOK"|"APPROVAL",
       "sourceRevision":int>=1,"mode":"CLEAN_UP"|"SUMMARIZE"|"CUSTOM","instruction":string|null (<=2000)}
  pre-stream problem+json: 400 | 401 | 404 DOCUMENT_NOT_FOUND | 409 REVISION_SUPERSEDED |
       413 {"reason":"TEXT_TOO_LARGE","characters":N,"maxCharacters":M}
  stream: event token {"text"} | event done {"finishReason":string|null,"truncated":bool,"complete":bool}
        | event error {"outcome":"OFFLINE"|"BUSY"|"MODEL_MISSING"|"DEADLINE"|"FAILED"|"TOO_LARGE",
                       "retryAfterSeconds":int|null,"characters":int|null,"maxCharacters":int|null}
        | ": hb"      exactly one done or error ends the stream; complete = finishReason=="stop" && !truncated
POST /api/assistant/documents/ai-revisions {"releaseId","documentType","sourceRevision","mode":"CLEAN_UP"|"CUSTOM",
       "instruction":string|null,"markdown":string (1..200000)} -> 201 {"revision":N+1}
  400 (incl. SUMMARIZE, NUL) | 404 | 409 REVISION_SUPERSEDED
```
