export type EnrichmentStatus = 'NOT_ATTEMPTED' | 'NO_LINK' | 'LINKED' | 'FAILED';

  targetBranch: string | null; countsTowardCommits: boolean };
export type CommitLink = { sha: string; shortSha: string; message: string; author: string | null; authoredAt: string; url: string };
