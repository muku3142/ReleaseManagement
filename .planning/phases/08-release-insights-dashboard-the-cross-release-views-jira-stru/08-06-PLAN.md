```ts
// Must equal DashboardController.MAX_HISTORY_SPAN_DAYS (DashboardSpanContractTest reads this line).
export const MAX_HISTORY_SPAN_DAYS = 366;
export const DEFAULT_HISTORY_DAYS = 30;          // D-08-13
export const HISTORY_PAGE_SIZE = 50;
```
- Types mirroring the `<interfaces>` JSON exactly: `DashboardSummary`, `DashboardReleaseRow`, `DashboardComposition`, `DashboardChurn`,
  `DashboardData { summary; stages: PromotionStage[]; releases: DashboardReleaseRow[] }`, `DeploymentEntry` (outcome typed
  `PromotionOutcome`), `DeploymentPage { entries; hasMore }`, `DashboardSearch { stage?: string; from?: string; to?: string; all?: boolean }`.
- `parseDashboardSearch(search: Record<string, unknown>): DashboardSearch` - `stage` kept only if it matches
  `/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i`; `from`/`to` kept only if `/^\d{4}-\d{2}-\d{2}$/` AND the date
  round-trips (`new Date(y, m - 1, d)` has the same y/m/d); `all` true for `true` or `'true'`, else undefined (UI-SPEC: malformed -> defaults).
- `defaultHistoryDates(today = new Date()): { from: string; to: string }` - local `yyyy-mm-dd`, `from` = today minus
  `DEFAULT_HISTORY_DAYS - 1` days.
- `historyRange(from: string, to: string): { ok: true; fromIso: string; toIso: string } | { ok: false; message: string }` -
  `start = new Date(y, m-1, d)`, `end = new Date(y2, m2-1, d2 + 1)`; `start > lastDay` -> `"Start date must be on or before end date."`;
  inclusive days `Math.round((end - start) / 86_400_000) > MAX_HISTORY_SPAN_DAYS` -> `"Choose a range of 366 days or less."` (build the
  message from the constant); else ISO strings of `start` and `end` (half-open, 08-RESEARCH A7: interpreted in the browser zone).
- `formatDuration(seconds: number | null): string` - null -> `'—'`; hours = s / 3600; >= 24 -> days with one decimal only when < 10
  (`Number(x.toFixed(1))`), else rounded, suffix `' d'`; below 24 -> hours, same rounding, suffix `' h'` (UI-SPEC Duration cell).
- `paperworkCounts(releaseIds: string[], stateOf: ReadinessLookup): { needPaperwork: number; stale: number }` - over `DOCUMENT_TYPES`,
  `state = stateOf(id, type) ?? 'NOT_GENERATED'`; needPaperwork counts releases with any `NOT_GENERATED` or `STALE` (ignoring
  `NO_TEMPLATE`); stale counts releases with any `STALE`. Comment: counts server-computed states, never re-derives staleness (T-05-55).
- `deploymentsPath(filter: { stageId?: string; fromIso: string; toIso: string }, offset: number): string` using `URLSearchParams`
  (`limit=HISTORY_PAGE_SIZE`).
- `mergeDeploymentPages(pages: DeploymentPage[]): DeploymentEntry[]` - concatenates, dropping ids already seen (OFFSET can shift when a
  promotion is recorded between loads, UI-SPEC).
- `nextDeploymentOffset(lastPage: DeploymentPage, pages: DeploymentPage[]): number | undefined` - total loaded rows when `hasMore`.
- Keys: `dashboardReleasesQueryKey = (all: boolean) => ['dashboard', 'releases', all ? 'all' : 'active'] as const`;
  `deploymentsQueryKey = (filter) => ['dashboard', 'deployments', filter.stageId ?? 'all', filter.fromIso, filter.toIso] as const`.
- Hooks: `useDashboardReleases(all: boolean)` = `useQuery({ queryKey, queryFn: () => apiFetch<DashboardData>(`/api/dashboard/releases?scope=${all ? 'all' : 'active'}`) })`;
  `useDeploymentHistory(filter | null)` = `useInfiniteQuery({ queryKey: deploymentsQueryKey(filter ?? placeholder), queryFn: ({ pageParam }) => apiFetch<DeploymentPage>(deploymentsPath(filter!, pageParam)), initialPageParam: 0, getNextPageParam: nextDeploymentOffset, enabled: filter !== null, placeholderData: keepPreviousData })`.
  `filter === null` means the typed range is invalid: no request is sent and the previous results stay (UI-SPEC).
Tests in `tests/api/dashboard.test.ts` per `<behavior>` (pure functions only; build ISO expectations with `new Date(y, m, d).toISOString()` so
they pass in any time zone).
