    `public void run(Plan plan, EventSink sink)`:
    ```
    start = System.nanoTime(); watchdog = timers.after(props.deadline(), () -> cancel.cancel(CancelReason.DEADLINE));
    heartbeat = timers.every(props.heartbeatInterval(), sink::heartbeat);
    try {
      sink.send(STATUS, new Status(requestId, "SEARCHING"));
      model = plan.collections().getFirst().embeddingModel();
      vector = embeddings.embed(model, EmbeddingPurpose.QUERY, List.of(question)).getFirst();   // not paused by D-10-03
      hits = chunks.search(ids, model, vector, plan.filter(), plan.k());
      topScore = hits.isEmpty() ? 0 : hits.getFirst().score();
      if (hits.isEmpty() || topScore < props.minScore()) -> refuse("GATE", null)   // no model call
      sources = group hits by (collectionId, documentId) in score order into SourceDocument, n = 1.. in first-seen order,
                collection = name from plan.collections(), topScore = first hit's score, passages = hit texts in order
      prompt = promptBuilder.chat(sources, plan.history(), question, props.numPredict());
      sink.send(STATUS, new Status(requestId, "GENERATING"));
      answer = new StringBuilder(); guard = new RefusalGuard(text -> { markFirstToken(); sink.send(TOKEN, new Token(text)); });
      completion = chat.stream(prompt.messages(), new GenerationOptions(props.numCtx(), props.numPredict(), props.temperature()),
                               token -> { answer.append(token); guard.accept(token); }, cancel);
      if (completion.isCancelled()) -> if DEADLINE send ERROR(new Error(ErrorReason.DEADLINE)); outcome = reason name; return
      guard.finish();
      truncated = calibration(prompt.estimatedTokens(), completion.promptEvalCount())    // see below
      if (guard.refused() || !guard.emittedAny()) -> refuse("MODEL", completion)
      else { result = citationBuilder.build(answer.toString(), prompt.sourcesUsed());
             sink.send(CITATIONS, new Citations(result.basis(), result.items()));
             sink.send(DONE, new Done(false, null, completion.finishReason(), truncated, completion.promptEvalCount(), completion.evalCount()));
             outcome = OK }
    } catch (RuntimeException e) {
      reason = cancel.reason() == CancelReason.DEADLINE ? ErrorReason.DEADLINE : errorReason(e);
      if (cancel.reason() != CancelReason.CLIENT_GONE) sink.send(ERROR, new Error(reason));
      outcome = reason or CLIENT_GONE; if INTERNAL: log.error("generation requestId={} failed with {}", requestId, e.getClass().getSimpleName());
    } finally {
      watchdog.cancel(false); heartbeat.cancel(false); gate.release(clientId);   // release BEFORE complete
      log metadata line; sink.complete();
    }
    ```
