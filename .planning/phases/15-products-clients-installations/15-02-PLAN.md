package com.evision.releasemgmt.domain.portfolio;
public record Product(UUID id, String name, Instant archivedAt) { public boolean archived() { return archivedAt != null; } }
public record Client(UUID id, String name, Instant archivedAt, Set<UUID> productIds) {
    public Client { productIds = Set.copyOf(productIds); }
    public boolean archived() { return archivedAt != null; }
    public boolean linkedTo(UUID productId) { return productIds.contains(productId); } }
public record ReleaseProduct(UUID id, String name, boolean archived, boolean hasInstalls) { }
public enum ClientDeploymentOutcome { SUCCESS, FAILED, ROLLED_BACK, VOIDED }
public record ClientDeployment(UUID id, UUID releaseId, UUID productId, UUID clientId, ClientDeploymentOutcome outcome,
        Instant installedAt, Instant recordedAt, String recordedBy, String note, UUID supersedesId, String correctionReason) { }
/** installedAt null means "now". */
public record ClientDeploymentDraft(UUID productId, ClientDeploymentOutcome outcome, Instant installedAt, String note) { }
public record RejectedClient(UUID clientId, String name, Reason reason) {   // name null when the id is unknown
    public enum Reason { NOT_LINKED, ARCHIVED } }
public record CurrentRun(UUID productId, UUID releaseId, String releaseName, String releaseVersion, Instant installedAt) { }
public record ClientInstall(ClientDeployment entry, String releaseName, String releaseVersion) { }
public enum ClientDeploymentRejection { INSTALL_RELEASE_NOT_FROZEN, INSTALL_PRODUCT_NOT_ON_RELEASE, INSTALL_PRODUCT_ARCHIVED,
        INSTALL_CLIENTS_REJECTED, INSTALL_ALREADY_SUPERSEDED, INSTALL_INSTALLED_AT_INVALID, INSTALL_OUTCOME_NOT_RECORDABLE }
public class ClientDeploymentRejectedException extends RuntimeException {   // reason() + clients() (List.of() unless CLIENTS_REJECTED)
public enum PortfolioEditRejection { PRODUCT_NAME_DUPLICATE, CLIENT_NAME_DUPLICATE, PRODUCT_ARCHIVED, CLIENT_ARCHIVED,
        CLIENT_LINK_HAS_INSTALLS, PRODUCT_HAS_INSTALLS_ON_RELEASE,
        RELEASE_PRODUCT_REQUIRED,    // D-15-18: create-release with no product
        RELEASE_LAST_PRODUCT }       // D-15-18: a product change would leave the release with none
public class PortfolioEditRejectedException extends RuntimeException { reason() }
ProductNotFoundException, ClientNotFoundException, ClientDeploymentNotFoundException  (RuntimeException, no-arg)

public record ClientDeploymentHistory(UUID releaseId, boolean frozen, List<ReleaseProduct> products, List<Client> clients,
                                      List<ClientDeployment> entries) {
    public static final Duration FUTURE_TOLERANCE = Duration.ofMinutes(2);
    public static final Comparator<ClientDeployment> LEDGER_ORDER;   // installedAt, then recordedAt
    public Optional<UUID> supersededBy(UUID deploymentId);
    public boolean isEffective(ClientDeployment d);
    public List<ClientDeployment> record(ClientDeploymentDraft draft, List<UUID> clientIds, String actor, Instant now);
    public ClientDeployment correct(UUID originalId, ClientDeploymentOutcome outcome, Instant installedAt, String note,
                                    String reason, String actor, Instant now);
}

package com.evision.releasemgmt.ports;
public interface ProductRepository {
    List<Product> findAll();                       // archived included, ordered by lower(name)
    Optional<Product> findById(UUID productId);
    Product add(String name);                      // PRODUCT_NAME_DUPLICATE on case-insensitive clash
    void rename(UUID productId, String name);      // PRODUCT_NAME_DUPLICATE; ProductNotFoundException
    void setArchivedAt(UUID productId, Instant archivedAt);   // null unarchives
    Map<UUID, Long> clientCounts();                // linked clients per product (archived clients included)
    Map<UUID, Long> releaseCounts();               // releases covering each product
}
public interface ClientRepository {
    List<Client> findAll();                        // archived included, ordered by lower(name), each with its productIds
    Optional<Client> findById(UUID clientId);
    Client add(String name);                       // CLIENT_NAME_DUPLICATE
    void rename(UUID clientId, String name);
    void setArchivedAt(UUID clientId, Instant archivedAt);
    void link(UUID clientId, UUID productId);      // idempotent
    void unlink(UUID clientId, UUID productId);    // CLIENT_LINK_HAS_INSTALLS when any client_deployment row exists (D-15-13/15)
}
public interface ReleaseProductRepository {
    List<ReleaseProduct> productsOf(UUID releaseId);                          // ordered by lower(name)
    Map<UUID, List<ReleaseProduct>> productsOfReleases(Collection<UUID> releaseIds);  // releases without products absent
    void replace(UUID releaseId, Set<UUID> productIds);   // D-15-01/02/07/15/18, see plan 04; empty set -> RELEASE_LAST_PRODUCT
}
// Header comment copied from PromotionRepository, renamed: append-only by construction, no update/delete/remove/save.
public interface ClientDeploymentRepository {
    ClientDeploymentHistory load(UUID releaseId);       // ReleaseNotFoundException when missing
    List<ClientDeployment> appendUnderLock(UUID releaseId, Function<ClientDeploymentHistory, List<ClientDeployment>> decide);
    List<ClientInstall> historyOfClient(UUID clientId); // every row, superseded and voided included
    List<CurrentRun> currentRunsOfClient(UUID clientId); // from current_client_release
}
