CREATE VIEW effective_client_deployment AS SELECT d.id, d.release_id, d.product_id, d.client_id, d.outcome, d.installed_at, d.recorded_at, d.recorded_by, d.note
  FROM client_deployment d WHERE d.outcome <> 'VOIDED' AND NOT EXISTS (SELECT 1 FROM client_deployment s WHERE s.supersedes_id = d.id);
