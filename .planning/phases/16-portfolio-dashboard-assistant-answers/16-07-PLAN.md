<interfaces>
From 16-04 (frontend/src/api/dashboard.ts): `usePortfolioSummary(productIds: string[], includeArchived: boolean)`, `usePortfolioBehind(...)`, `usePortfolioMatrix(...)`
returning React Query results of `PortfolioSummary`, `PortfolioBehind`, `PortfolioMatrix`:
```ts
type ReleaseBrief = { releaseId: string; name: string; version: string; frozenAt: string };
type RunBrief = { releaseId: string; name: string; version: string; installedAt: string; inProduction: boolean };
type StatusCounts = { open: number; frozen: number; inProduction: number };
type PortfolioProductRow = { id; name; archived; releases: StatusCounts; newestFrozen: ReleaseBrief | null; linkedClients: number; clientsOnNewest: number };
type PortfolioTiles = { clientsBehind; notInstalled; activeProducts; activeClients; installsLast30Days };
type PortfolioSummary = { tiles: PortfolioTiles; products: PortfolioProductRow[]; totals: StatusCounts; unassigned: StatusCounts | null };
type BehindRow = { clientId; clientName; clientArchived; productId; productName; runs: RunBrief; releasesBehind: number; newest: ReleaseBrief };
type NotInstalledRow = { clientId; clientName; clientArchived; productId; productName; newest: ReleaseBrief | null };
type PortfolioBehind = { behind: BehindRow[]; notInstalled: NotInstalledRow[] };
type MatrixColumn = { id; name; archived };
type MatrixCell = { clientId; productId; run: RunBrief | null; releasesBehind: number | null; newest: ReleaseBrief | null; upToDate: boolean };
type PortfolioMatrix = { products: MatrixColumn[]; clients: MatrixColumn[]; cells: MatrixCell[] };  // cells = linked pairs only
```
Server already sorts rows (names case-insensitive, archived last; behind by N desc) and applies archived/product scope.
