package com.evision.releasemgmt.domain.briefing;
public record BriefingIssue(String key, String summary, String issueType, String status, StatusCategory statusCategory,
        String priority, boolean flagged, String assignee, Instant created, Instant updated, Instant resolved,
        LocalDate dueDate, BigDecimal storyPoints, String parentKey, int subtaskCount,
        boolean acceptanceCriteriaPresent, String description, List<Sprint> sprints) {
    public record Sprint(long id, String name, String state, Instant startDate, Instant endDate) { }
}
// Resolved custom field ids for the site; null = the site has no such field (rule reads "not checked").
public record BriefingFieldIds(String storyPoints, String flagged, String sprint, String acceptanceCriteria) { }
public record BriefingContent(BriefingStatus status, int attention, String headline, String summary, List<Section> sections) {
    public record Section(String title, List<Item> items) { }
    public record Item(String issueKey, String text) { }   // issueKey null for non-issue lines
}
public record BriefingThresholds(int staleAfterWorkingDays, int dueSoonDays, int backlogItems) {
    public static final BriefingThresholds DEFAULTS = new BriefingThresholds(3, 5, 15); }
public record BriefingRunSettings(BriefingThresholds thresholds, WorkCalendar calendar, int keepRuns) {
    public static final int DEFAULT_KEEP_RUNS = 60; }
public record WorkCalendar(ZoneId zone, Set<DayOfWeek> workDays) {
    public static final ZoneId CAIRO = ZoneId.of("Africa/Cairo");
    public static final Set<DayOfWeek> SUN_TO_THU = EnumSet.of(SUNDAY, MONDAY, TUESDAY, WEDNESDAY, THURSDAY);
    public LocalDate today(Instant now);
    public int workingDaysElapsed(Instant from, Instant now);   // work days after from's Cairo date through now's Cairo date inclusive
    public Instant weekStart(Instant now);                      // previous-or-same Sunday 00:00 in zone
    public Instant previousWorkingDayStart(Instant now);        // 00:00 of the latest work day strictly before today
}
public final class BriefingQueries {
    public static final Pattern PROJECT_KEY = Pattern.compile("^[A-Z][A-Z0-9_]{1,9}$");
    public static final Pattern ISSUE_KEY = Pattern.compile("^[A-Z][A-Z0-9_]{1,9}-\\d+$");
    public static final int UNCAPPED = Integer.MAX_VALUE;
    public static String activeSprintOpen(Collection<String> keys);       // project in ("A","B") AND sprint in openSprints() AND statusCategory != Done ORDER BY Rank ASC
    public static String activeSprintAll(Collection<String> keys);        // project in (...) AND sprint in openSprints()
    public static String nextSprint(Collection<String> keys);             // ... AND sprint in futureSprints() AND statusCategory != Done ORDER BY Rank ASC
    public static String backlog(Collection<String> keys);                // ... AND statusCategory != Done AND (sprint is EMPTY OR sprint not in (openSprints(), futureSprints())) ORDER BY Rank ASC
    public static String staleCandidates(Collection<String> keys, int staleAfterWorkingDays);   // ... AND statusCategory = "In Progress" AND updated <= -{n}d
    public static String inProgress(Collection<String> keys);             // ... AND statusCategory = "In Progress"
    public static String createdSince(Collection<String> keys, int daysBack);                   // ... AND created >= -{n}d
    public static String doneSince(Collection<String> keys, int daysBack);                      // ... AND statusCategory = Done AND resolutiondate >= -{n}d
    public static String reopenCandidates(Collection<String> keys, int daysBack);               // ... AND statusCategory != Done AND updated >= -{n}d ORDER BY updated DESC
    public static String dueWithin(Collection<String> keys, int days);    // ... AND statusCategory != Done AND duedate <= {n}d
    public static int daysBack(Instant since, Instant now);               // ceil(hours/24) + 1, clamped to 1..30
    // Any invalid or empty key collection -> IllegalArgumentException (keys come from the DB CHECK, so this is defence in depth)
}
public final class BriefingText {   // shared by all engines
    public static String clip(String s, int maxCodePoints);   // null -> "", strip, cut by code points, appends "…" when cut
    public static boolean isBlocker(BriefingIssue i);          // flagged || priority in {Highest, Blocker} (case-insensitive) || status equalsIgnoreCase "Blocked"
    public static String blockerReason(BriefingIssue i);       // "flagged" | "priority Highest" | "status Blocked" (first that applies)
    public static boolean isStory(BriefingIssue i);            // issueType equalsIgnoreCase "Story"
}
public record DigestInput(String productName, List<BriefingIssue> activeSprintOpen, List<BriefingIssue> staleCandidates,
        List<BriefingIssue> nextSprint, List<BriefingIssue> created, List<BriefingIssue> done, List<BriefingIssue> reopened,
        List<BriefingIssue> dueCandidates, Instant since, Instant now, BriefingRunSettings settings, BriefingFieldIds fields) { }
public final class DigestRules { public static BriefingContent compute(DigestInput in); }

package com.evision.releasemgmt.ports;
public interface JiraBriefingPort {
    // Read-only POST /search/jql with cursor paging; returns at most maxIssues (BriefingQueries.UNCAPPED = no cap, the max-pages guard applies).
    // Failures throw JiraTicketReadException with a JiraAuthFailureReason (INVALID_CREDENTIALS on 401).
    List<BriefingIssue> search(JiraAccessContext access, String jql, BriefingFieldIds fieldIds, int maxIssues);
}

    - Stale (from staleCandidates): statusCategory IN_PROGRESS and workingDaysElapsed(updated, now) >= 3 -> "{summary}: no update for {n} working days ({assignee or 'Unassigned'})"; 2 working days -> not listed even though JQL prefilter returned it
    - Not ready (stories in activeSprintOpen ∪ nextSprint, de-duplicated by key): missing acceptance criteria, estimate (storyPoints null) or epic (parentKey null) -> "{summary}: missing acceptance criteria, estimate, epic" listing only the missing ones; with `fields.storyPoints() == null` the estimate is never reported missing; non-stories ignored
    - Changes since the last run: created with created >= since -> "{summary}: created"; done with resolved >= since -> "{summary}: moved to Done"; reopened -> "{summary}: reopened"; items older than since are excluded although the JQL prefilter returned them
    - Due soon or overdue (from dueCandidates, statusCategory != DONE): dueDate < today -> "{summary}: overdue since {yyyy-MM-dd}"; today <= dueDate <= today+5 -> "{summary}: due {yyyy-MM-dd}"; later -> excluded
    - Sections in this order with these titles: "Blockers", "Stale in progress", "Not ready", "Changes since the last run", "Due soon or overdue"; items keep issueKey
    - attention = |blockers| + |stale| + |notReady| + |due| (an issue counts once per section); status ATTENTION when attention > 0; QUIET when every section is empty; else OK
    - headline (<= 140): attention > 0 -> "{b} blockers, {s} stale, {n} not ready, {d} due soon or overdue in CNV"; otherwise "Nothing needs attention in CNV"
    - summary: line 1 counts incl. changes "{c} changes since the last run"; line 2 "Most important: {KEY} {summary}" (first blocker, else first overdue, else first stale) or omitted; joined with "\n", <= 400
    - Item text is clipped to 300 code points
