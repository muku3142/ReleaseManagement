package com.evision.releasemgmt.domain.briefing;
public enum BriefingTrigger { SLOT, MANUAL }
public record AppBriefingRun(UUID id, BriefingTask task, UUID productId, Instant slotAt, BriefingTrigger trigger,
        BriefingStatus status, String reason, Integer attention, String headline, String summary,
        List<BriefingContent.Section> sections, Instant startedAt, Instant finishedAt) {
    public static final String REASON_CLAUDE_COVERED = "CLAUDE_COVERED";
    public static final String REASON_NO_JIRA_PROJECT = "NO_JIRA_PROJECT";
}
public record BriefingSettings(String schedulerAccountId, int graceMinutes, Set<DayOfWeek> workDays, Integer keepRuns,
        Instant updatedAt, String updatedBy) { }
public record BriefingTaskSettings(BriefingTask task, boolean enabled, String cron,
        Integer staleAfterWorkingDays, Integer dueSoonDays, Integer backlogItems) { }   // null threshold = use config.json / default (D-17.1-16)

package com.evision.releasemgmt.ports;
public interface BriefingRunRepository {
    /** False when (task, product, slotAt) already has a run; never throws for that case. */
    boolean insert(AppBriefingRun run);
    boolean existsForSlot(BriefingTask task, UUID productId, Instant slotAt);
    /** finished_at of the newest ok/attention/quiet run of this task and product. */
    Optional<Instant> lastSuccessfulFinishedAt(BriefingTask task, UUID productId);
    /** All runs of a task, newest first by finished_at. */
    List<AppBriefingRun> findByTask(BriefingTask task);
    /** Keeps the newest keepRuns runs of (task, product) by finished_at; deletes the rest in one statement. */
    void prune(BriefingTask task, UUID productId, int keepRuns);
    /** Status of the newest non-skipped run across all tasks and products. */
    Optional<BriefingStatus> latestStatus();
}
public interface BriefingSettingsRepository {
    BriefingSettings get();
    /** The three app-fallback tasks in BriefingTask order. */
    List<BriefingTaskSettings> tasks();
    /** Never changes scheduler_account_id. */
    void saveSchedule(int graceMinutes, Set<DayOfWeek> workDays, Integer keepRuns, List<BriefingTaskSettings> tasks, Instant at, String by);
    /** Stores only the account id (no credential read, no decryption). */
    void setSchedulerAccount(String accountId, Instant at);
}
