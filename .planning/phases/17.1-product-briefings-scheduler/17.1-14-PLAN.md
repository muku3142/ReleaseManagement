final class BriefingSlots {   // application/briefing
    static final Duration CLAUDE_EARLY_TOLERANCE = Duration.ofMinutes(5);
    // Latest cron fire time <= cutoff, searching from cutoff - lookback; empty when none.
    static Optional<Instant> latestSlotAtOrBefore(String cron, ZoneId zone, Instant cutoff, Duration lookback);
    static boolean claudeCovers(BriefingFeed feed, Instant slot, Duration grace);   // !readError && any run: status != FAILED && runAt in [slot-5m, slot+grace]
}
@Service public class BriefingScheduleService {
    @Scheduled(initialDelayString = "${app.briefings.initial-delay}", fixedDelayString = "${app.briefings.tick-interval}")
    public void scheduledTick();          // tick(clock.instant())
    public void tick(Instant now);        // public for tests and 17.1-16
}
