```ts
// features.ts
export const dataTableFeatures = tableFeatures({ ... });   // the single extension point; 23.1 adds row selection/expanding/pinning here
export type DataTableFeatures = typeof dataTableFeatures;
// DataTable.tsx
export type FacetedFilter = { columnId: string; title: string; options: { value: string; label: string; icon?: LucideIcon }[] };
export function DataTable<TData>(props: {
  columns: ColumnDef<DataTableFeatures, TData>[];   // v9 column defs; build with createColumnHelper<DataTableFeatures, TData>()
  data: TData[];
  getRowId: (row: TData) => string;
  caption: string;               // required, rendered sr-only, labels the scroll region
  noun: string;                  // plural, e.g. "releases"
  toolbar?: React.ReactNode;     // page slot left of the facets
  facetedFilters?: FacetedFilter[];
  emptyState?: React.ReactNode;  // shown when data is empty
  isLoading?: boolean;           // header + 5 skeleton rows
  stickyFirstColumn?: boolean;
}): JSX.Element;
// DataTableColumnHeader.tsx
export function DataTableColumnHeader(props: { column: Column<DataTableFeatures, any>; title: string; className?: string }): JSX.Element;
```
Discretion recorded: the UI-SPEC API lists a `features` prop. The component instead always uses the shared `dataTableFeatures`
instance, because v9 column types are parameterised by the features type and RESEARCH Pattern 8 makes that one object the extension
point; a per-call prop would add generic plumbing with no caller that needs a different set (CLAUDE.md: no speculative abstraction).
Column meta used by the table: `meta: { headerClassName?, cellClassName? }` for right-aligned numeric columns, and `enableHiding: false` for the identity column.
