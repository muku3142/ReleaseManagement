```ts
// src/hooks/useCountUp.ts
// Returns the number to SHOW while counting (0 -> value over durationMs, ease-out), or null when nothing is animating.
// Animates only the first time `value` becomes a finite number > 0; never under reduced motion; never again after that.
export function useCountUp(value: number | null, durationMs?: number /* default 600 */): number | null;

// src/hooks/usePulseOnChange.ts
// pulsing becomes true when `value` changes after the first render (never on mount, never under reduced motion);
// spread onAnimationEnd on the pulsing element so it resets.
export function usePulseOnChange<T>(value: T): { pulsing: boolean; onAnimationEnd: () => void };

// src/hooks/useEntranceStagger.ts
// Pass ready=true once real rows are rendered. Returns per-index props only for the first ready render; {} afterwards
// (re-sort, filter, refetch) and always {} under reduced motion.
export function useEntranceStagger(ready: boolean): (index: number) => { className?: string; style?: React.CSSProperties };

// Card: new optional prop
interactive?: boolean;
// DataTable: new optional prop
onRowClick?: (row: TData) => void;
// ProductOpsMark and ProductOpsLogo: new optional prop
animate?: boolean;
```
